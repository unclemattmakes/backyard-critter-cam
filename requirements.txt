# Backyard Critter Cam -- runtime dependencies (built/tested on Python 3.14, Windows).
#
# SUPPORTED INSTALL PATH: run `setup.bat` (Windows) or `bash setup.sh` (Linux/macOS). Those
# scripts pick the right torch build for your hardware -- the cu130 wheels below on an NVIDIA
# GPU, a CPU-only torch otherwise -- and then install this file. Do that instead of a bare
# `pip install -r requirements.txt`, which does NOT install torch (see the next paragraph).
#
# torch + torchvision are installed SEPARATELY (see README "Setup" step 2) because the right
# build depends on the GPU. Blackwell cards (RTX 50-series) are compute capability sm_120, and
# only a CUDA 12.8+ build ships sm_120 kernels; the stock cu126 wheels a plain `pip install
# torch` pulls fail on them with "no kernel image is available for execution on the device"
# (torch.cuda.is_available() still returns True -- the first real op is what dies). On such a
# card install the cu130 wheels:
#
#   pip install torch==2.12.0 torchvision==0.27.0 --index-url https://download.pytorch.org/whl/cu130
#
# (Older NVIDIA GPUs and CPU-only boxes use the default wheels -- setup.bat / setup.sh pick for
# you. This file intentionally does NOT pin torch, so it stays correct on every machine.)
#
# Then install the rest from this file. (ultralytics pulls numpy/pillow/etc.; it will reuse
# the cu130 torch already present rather than replacing it.)

opencv-python>=4.10,<5
# ultralytics>=8.4 deliberately excludes 8.3.41/8.3.42 -- the Dec-2024 PyPI releases that shipped a
# cryptominer -- and <9 stops a future major auto-installing. For a stronger guarantee, generate a
# hashed lockfile (pip-compile --generate-hashes) and install with --require-hashes.
ultralytics>=8.4,<9
# numpy and pillow are imported DIRECTLY all over this codebase (embed.py, reid.py, eval.py,
# import_trailcam.py, ...), so they are declared directly -- ultralytics/pybioclip happen to pull
# them in today, but an upstream dropping a transitive dep must not break this install.
numpy>=1.26,<3
pillow>=10,<12
# The general-CLIP "is this even an animal?" gate (clipfilter.py) imports open_clip directly;
# declared in its own right rather than ridden in as a pybioclip dependency.
open_clip_torch>=2.24,<3
# Sun-driven day/night camera profiles (daynight.py), and the batch's own start time
# (sunsched.py aims it at the late-afternoon glare window, sunset-2.5h). Without it the rig still
# runs, but current_period() can't resolve dawn/dusk and the profiles disable (camera just
# auto-exposes), and the batch keeps whatever start time it last had.
astral>=3.2,<4
# Process inspection for the self-healing pair: rigwatch.py asks whether backyard_cam.py is still
# alive before restarting it, and heavyio.py uses it to reap a lock whose owner died and to watch
# the cloud client's io counters. NOT optional for rigwatch -- without it rig_pids() cannot tell a
# live rig from a dead one, and it refuses to act rather than restart a healthy rig every 5 min.
psutil>=5.9,<8
# Species classifier (classify.py): BioCLIP 2 zero-shot. NOTE the PyPI distribution is named
# `pybioclip` but it imports as `bioclip`. Reuses the separately-installed cu130 torch (header).
pybioclip>=2.1,<3
# Phase 3 re-identification. embed.py loads MegaDescriptor (the wildlife re-ID foundation model)
# through timm; reid.py clusters the resulting vectors with scipy's hierarchical clustering.
# Both reuse the cu130 torch / numpy / pillow already present (timm is also a pybioclip dep).
timm>=1.0,<2
scipy>=1.13,<2

# --- Development only (not needed to run the rig) ---
# Test suite: `python -m pytest tests/ -q`. Pure-logic tests, no GPU/camera/model needed.
pytest>=8.0,<10
